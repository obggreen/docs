## GET /v1/chatgpt-images/generations/{task_uuid}

Check the current status of your image generation task.

<ParamField path="task_uuid" type="string" required>
  Task UUID returned from the generate endpoint
</ParamField>

<Accordion title="Status Values">
  <ResponseField name="pending" type="string">
    Task is in queue waiting to be processed
  </ResponseField>
  <ResponseField name="processing" type="string">
    Image is currently being generated
  </ResponseField>
  <ResponseField name="completed" type="string">
    Image generation completed successfully
  </ResponseField>
  <ResponseField name="failed" type="string">
    Image generation failed with error
  </ResponseField>
</Accordion>
