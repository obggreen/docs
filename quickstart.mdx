### Step 2: Make Your First Request

Choose your preferred language and start generating:

<CodeGroup>

```python Python
import requests

url = "https://api.runblob.com/v1/video/veo31/generate"
headers = {
    "Authorization": "Bearer YOUR_API_KEY",
    "Content-Type": "application/json"
}
payload = {
    "prompt": "A serene mountain landscape at sunset with flowing clouds",
    "duration": 5
}

response = requests.post(url, json=payload, headers=headers)
result = response.json()
print(f"Video URL: {result['video_url']}")
```


```javascript JavaScript
const response = await fetch('https://api.runblob.com/v1/video/veo31/generate', {
  method: 'POST',
  headers: {
    'Authorization': 'Bearer YOUR_API_KEY',
    'Content-Type': 'application/json'
  },
  body: JSON.stringify({
    prompt: 'A serene mountain landscape at sunset with flowing clouds',
    duration: 5
  })
});

const result = await response.json();
console.log(`Video URL: ${result.video_url}`);
```


```bash cURL
curl -X POST https://api.runblob.com/v1/video/veo31/generate \
  -H "Authorization: Bearer YOUR_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{
    "prompt": "A serene mountain landscape at sunset with flowing clouds",
    "duration": 5
  }'
```


```go Go
package main

import (
    "bytes"
    "encoding/json"
    "fmt"
    "net/http"
)

func main() {
    url := "https://api.runblob.com/v1/video/veo31/generate"
    payload := map[string]interface{}{
        "prompt":   "A serene mountain landscape at sunset with flowing clouds",
        "duration": 5,
    }
    
    jsonData, _ := json.Marshal(payload)
    req, _ := http.NewRequest("POST", url, bytes.NewBuffer(jsonData))
    req.Header.Set("Authorization", "Bearer YOUR_API_KEY")
    req.Header.Set("Content-Type", "application/json")
    
    client := &http.Client{}
    resp, _ := client.Do(req)
    defer resp.Body.Close()
    
    var result map[string]interface{}
    json.NewDecoder(resp.Body).Decode(&result)
    fmt.Printf("Video URL: %s\n", result["video_url"])
}
```

</CodeGroup>

### Step 3: Check Status & Retrieve Video

Video generation is asynchronous. Use the task ID to check status:

<CodeGroup>

```python Python
import time

task_id = result['task_id']
status_url = f"https://api.runblob.com/v1/tasks/{task_id}"

while True:
    status_response = requests.get(status_url, headers=headers)
    status_data = status_response.json()
    
    if status_data['status'] == 'completed':
        print(f"✅ Video ready: {status_data['video_url']}")
        break
    elif status_data['status'] == 'failed':
        print(f"❌ Generation failed: {status_data['error']}")
        break
    
    print(f"⏳ Status: {status_data['status']} ({status_data['progress']}%)")
    time.sleep(5)
```


```javascript JavaScript
const taskId = result.task_id;
const statusUrl = `https://api.runblob.com/v1/tasks/${taskId}`;

async function checkStatus() {
  while (true) {
    const statusResponse = await fetch(statusUrl, {
      headers: { 'Authorization': 'Bearer YOUR_API_KEY' }
    });
    const statusData = await statusResponse.json();
    
    if (statusData.status === 'completed') {
      console.log(`✅ Video ready: ${statusData.video_url}`);
      break;
    } else if (statusData.status === 'failed') {
      console.log(`❌ Generation failed: ${statusData.error}`);
      break;
    }
    
    console.log(`⏳ Status: ${statusData.status} (${statusData.progress}%)`);
    await new Promise(resolve => setTimeout(resolve, 5000));
  }
}

checkStatus();
```

</CodeGroup>
